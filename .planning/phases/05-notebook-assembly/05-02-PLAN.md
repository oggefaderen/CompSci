    def get_src(cell):
        return ''.join(cell['source'])

    def set_src(cell, new_src):
        cell['source'] = new_src

    # --- Fix 1: Country assortativity 100-network loop ---
    # Find by: contains "country_rs_random" and "for i in tqdm"
    for i, cell in enumerate(cells):

            lines = src.split('\n')
            new_lines = []
            removed = []
            for line in lines:
                # Remove only lines that are bare print() calls inside the loop body (4+ space indent)
                if re.match(r'^\s{4,}print\(', line):
                    removed.append(line.strip())
                else:
                    new_lines.append(line)
            set_src(cell, '\n'.join(new_lines))
            print(f'Country loop cell [{i}]: removed {len(removed)} indented print lines: {removed}')
            break

    # --- Fix 2: Degree assortativity 100-network loop ---
    # Find by: contains "degree_rs_random" and "for i in tqdm"
    for i, cell in enumerate(cells):
        src = get_src(cell)
        if 'degree_rs_random' in src and 'for i in tqdm' in src and cell['cell_type'] == 'code':
            lines = src.split('\n')
            new_lines = []
            removed = []
            for line in lines:
                if re.match(r'^\s{4,}print\(', line):
                    removed.append(line.strip())
                else:
                    new_lines.append(line)
            set_src(cell, '\n'.join(new_lines))
            print(f'Degree loop cell [{i}]: removed {len(removed)} indented print lines: {removed}')
            break

    # --- Verify Fix 3: Q4 TF-IDF display cell already uses [:10] ---
    # Find by: contains "top_tf_words" and "top_tfidf_words"
    for i, cell in enumerate(cells):
        src = get_src(cell)
        if 'top_tf_words' in src and 'top_tfidf_words' in src and cell['cell_type'] == 'code':
            if '[:10]' in src:
                print(f'Q4 TF-IDF cell [{i}]: [:10] slice confirmed, no change needed.')
            else:
                # If somehow missing, fix it
                new_src = re.sub(r'top_tf_words = sorted\([^)]+\)(?!\[:)',
                                 lambda m: m.group(0) + '[:10]', src)
                new_src = re.sub(r'top_tfidf_words = sorted\([^)]+\)(?!\[:)',
                                 lambda m: m.group(0) + '[:10]', new_src)
                set_src(cell, new_src)
                print(f'Q4 TF-IDF cell [{i}]: added [:10] slices.')
            break

    # --- Clear all saved outputs for fresh Kernel > Restart & Run All ---
    cleared = 0
    for cell in cells:
        if cell['cell_type'] == 'code':
            if cell.get('outputs'):
                cell['outputs'] = []
                cleared += 1
            cell['execution_count'] = None
    print(f'Cleared outputs from {cleared} code cells.')
